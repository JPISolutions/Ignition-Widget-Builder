import { useState } from 'react';

// Simulated live data - replace with actual data binding
const mockData = {
  state: { name: 'Running', type: 'production' },
  oee: 84.2,
  availability: 92.1,
  performance: 95.8,
  quality: 95.4,
  rateActual: 245,
  rateStandard: 260,
  countInfeed: 12450,
  countOutfeed: 12380,
  countDefect: 70
};

const stateColors = {
  production: { bg: 'bg-emerald-500', text: 'text-emerald-500', glow: 'shadow-emerald-500/50' },
  idle: { bg: 'bg-amber-500', text: 'text-amber-500', glow: 'shadow-amber-500/50' },
  down: { bg: 'bg-red-500', text: 'text-red-500', glow: 'shadow-red-500/50' },
  planned: { bg: 'bg-blue-500', text: 'text-blue-500', glow: 'shadow-blue-500/50' }
};

const getOeeColor = (val) => {
  if (val >= 85) return 'text-emerald-400';
  if (val >= 70) return 'text-amber-400';
  return 'text-red-400';
};

const MiniGauge = ({ value, label, color = 'emerald' }) => (
  <div className="flex flex-col items-center">
    <div className="relative w-12 h-12">
      <svg className="w-12 h-12 -rotate-90">
        <circle cx="24" cy="24" r="20" fill="none" stroke="#374151" strokeWidth="4" />
        <circle 
          cx="24" cy="24" r="20" fill="none" 
          stroke={`url(#gradient-${label})`}
          strokeWidth="4" 
          strokeLinecap="round"
          strokeDasharray={`${value * 1.256} 125.6`}
        />
        <defs>
          <linearGradient id={`gradient-${label}`} x1="0%" y1="0%" x2="100%" y2="0%">
            <stop offset="0%" stopColor={value >= 85 ? '#10b981' : value >= 70 ? '#f59e0b' : '#ef4444'} />
            <stop offset="100%" stopColor={value >= 85 ? '#34d399' : value >= 70 ? '#fbbf24' : '#f87171'} />
          </linearGradient>
        </defs>
      </svg>
      <span className="absolute inset-0 flex items-center justify-center text-xs font-semibold text-white">
        {value.toFixed(0)}%
      </span>
    </div>
    <span className="text-[10px] text-gray-400 mt-1 uppercase tracking-wide">{label}</span>
  </div>
);

export default function CapLoaderWidget() {
  const [data] = useState(mockData);
  const stateStyle = stateColors[data.state.type] || stateColors.idle;
  const ratePercent = ((data.rateActual / data.rateStandard) * 100).toFixed(0);

  return (
    <div className="w-80 bg-gradient-to-br from-gray-900 via-gray-800 to-gray-900 rounded-xl p-4 border border-gray-700/50 shadow-xl">
      {/* Header */}
      <div className="flex items-center justify-between mb-3">
        <div className="flex items-center gap-2">
          <div className={`w-2.5 h-2.5 rounded-full ${stateStyle.bg} shadow-lg ${stateStyle.glow} animate-pulse`} />
          <h3 className="text-white font-semibold text-sm">Cap Loader</h3>
        </div>
        <span className={`text-xs px-2 py-0.5 rounded-full bg-gray-700/50 ${stateStyle.text} font-medium`}>
          {data.state.name}
        </span>
      </div>

      {/* OEE Main Display */}
      <div className="flex items-center justify-between mb-4">
        <div>
          <div className="text-gray-400 text-xs uppercase tracking-wider mb-1">OEE</div>
          <div className={`text-4xl font-bold ${getOeeColor(data.oee)}`}>
            {data.oee.toFixed(1)}
            <span className="text-lg text-gray-500">%</span>
          </div>
        </div>
        
        {/* Mini Gauges */}
        <div className="flex gap-3">
          <MiniGauge value={data.availability} label="Avail" />
          <MiniGauge value={data.performance} label="Perf" />
          <MiniGauge value={data.quality} label="Qual" />
        </div>
      </div>
      {/* Rate Bar */}
      <div className="mb-3">
        <div className="flex justify-between text-xs mb-1">
          <span className="text-gray-400">Rate</span>
          <span className="text-white font-medium">{data.rateActual} / {data.rateStandard} <span className="text-gray-500">ppm</span></span>
        </div>
        <div className="h-1.5 bg-gray-700 rounded-full overflow-hidden">
          <div 
            className="h-full bg-gradient-to-r from-cyan-500 to-blue-500 rounded-full transition-all"
            style={{ width: `${Math.min(ratePercent, 100)}%` }}
          />
        </div>
      </div>

      {/* Counts Row */}
      <div className="grid grid-cols-3 gap-2 text-center">
        <div className="bg-gray-800/50 rounded-lg py-2 px-1">
          <div className="text-white font-semibold text-sm">{data.countInfeed.toLocaleString()}</div>
          <div className="text-gray-500 text-[10px] uppercase">Infeed</div>
        </div>
        <div className="bg-gray-800/50 rounded-lg py-2 px-1">
          <div className="text-white font-semibold text-sm">{data.countOutfeed.toLocaleString()}</div>
          <div className="text-gray-500 text-[10px] uppercase">Outfeed</div>
        </div>
        <div className="bg-gray-800/50 rounded-lg py-2 px-1">
          <div className="text-red-400 font-semibold text-sm">{data.countDefect}</div>
          <div className="text-gray-500 text-[10px] uppercase">Defects</div>
        </div>
      </div>
    </div>
  );
}
